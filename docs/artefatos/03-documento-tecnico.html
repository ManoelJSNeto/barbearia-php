Dockerfile

Imagem baseada em php:8.2-apache com as extensões mínimas necessárias.

# docker/Dockerfile
FROM php:8.2-apache

# extensões necessárias
RUN docker-php-ext-install pdo pdo_mysql mysqli

# habilitar mod_rewrite (para URLs limpas futuramente)
RUN a2enmod rewrite

# PHPMailer via composer
RUN apt-get update && apt-get install -y \
    unzip curl git \
  && curl -sS https://getcomposer.org/installer | php \
  && mv composer.phar /usr/local/bin/composer

# GD para redimensionar imagens do portfólio
RUN apt-get install -y libpng-dev libjpeg-dev libwebp-dev \
  && docker-php-ext-configure gd --with-jpeg --with-webp \
  && docker-php-ext-install gd

# configuração de upload
RUN echo "upload_max_filesize=10M" >> /usr/local/etc/php/php.ini \
 && echo "post_max_size=12M"       >> /usr/local/etc/php/php.ini \
 && echo "memory_limit=128M"       >> /usr/local/etc/php/php.ini

# diretório raiz do Apache
ENV APACHE_DOCUMENT_ROOT /var/www/html
RUN sed -ri 's!/var/www/html!${APACHE_DOCUMENT_ROOT}!g' \
    /etc/apache2/sites-available/*.conf

WORKDIR /var/www/html

Schema SQL completo

-- docker/mysql/init.sql
SET NAMES utf8mb4;
SET time_zone = '-03:00';

-- ── USUÁRIOS (clientes, barbeiros e admins) ──────────────────
CREATE TABLE usuarios (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  nome          VARCHAR(100) NOT NULL,
  email         VARCHAR(150) NOT NULL UNIQUE,
  senha_hash    VARCHAR(255) NOT NULL,
  perfil        ENUM('cliente','barbeiro','admin') NOT NULL DEFAULT 'cliente',
  ativo         TINYINT(1) NOT NULL DEFAULT 1,
  criado_em     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── TOKENS DE RECUPERAÇÃO DE SENHA ───────────────────────────
CREATE TABLE tokens_senha (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  usuario_id    INT UNSIGNED NOT NULL,
  token         VARCHAR(64) NOT NULL UNIQUE,
  expira_em     DATETIME NOT NULL,
  usado         TINYINT(1) NOT NULL DEFAULT 0,
  criado_em     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── CATEGORIAS DE SERVIÇO ─────────────────────────────────────
CREATE TABLE categorias (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  nome          VARCHAR(60) NOT NULL,
  descricao     VARCHAR(255),
  ordem         TINYINT UNSIGNED NOT NULL DEFAULT 0
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── SERVIÇOS ──────────────────────────────────────────────────
CREATE TABLE servicos (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  categoria_id  INT UNSIGNED NOT NULL,
  nome          VARCHAR(100) NOT NULL,
  descricao     TEXT,
  preco         DECIMAL(8,2) NOT NULL,
  duracao_min   SMALLINT UNSIGNED NOT NULL,  -- duração em minutos
  ativo         TINYINT(1) NOT NULL DEFAULT 1,
  criado_em     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (categoria_id) REFERENCES categorias(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── FOTOS DOS SERVIÇOS ────────────────────────────────────────
CREATE TABLE servico_fotos (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  servico_id    INT UNSIGNED NOT NULL,
  foto_path     VARCHAR(255) NOT NULL,
  ordem         TINYINT UNSIGNED NOT NULL DEFAULT 0,
  FOREIGN KEY (servico_id) REFERENCES servicos(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── COMBOS ────────────────────────────────────────────────────
CREATE TABLE combos (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  nome          VARCHAR(100) NOT NULL,
  descricao     TEXT,
  preco         DECIMAL(8,2) NOT NULL,
  duracao_min   SMALLINT UNSIGNED NOT NULL,
  ativo         TINYINT(1) NOT NULL DEFAULT 1,
  criado_em     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE combo_servicos (
  combo_id      INT UNSIGNED NOT NULL,
  servico_id    INT UNSIGNED NOT NULL,
  PRIMARY KEY (combo_id, servico_id),
  FOREIGN KEY (combo_id)   REFERENCES combos(id)   ON DELETE CASCADE,
  FOREIGN KEY (servico_id) REFERENCES servicos(id)  ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── SERVIÇOS QUE CADA BARBEIRO OFERECE ───────────────────────
CREATE TABLE barbeiro_servicos (
  barbeiro_id   INT UNSIGNED NOT NULL,
  servico_id    INT UNSIGNED NOT NULL,
  PRIMARY KEY (barbeiro_id, servico_id),
  FOREIGN KEY (barbeiro_id) REFERENCES usuarios(id) ON DELETE CASCADE,
  FOREIGN KEY (servico_id)  REFERENCES servicos(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE barbeiro_combos (
  barbeiro_id   INT UNSIGNED NOT NULL,
  combo_id      INT UNSIGNED NOT NULL,
  PRIMARY KEY (barbeiro_id, combo_id),
  FOREIGN KEY (barbeiro_id) REFERENCES usuarios(id) ON DELETE CASCADE,
  FOREIGN KEY (combo_id)    REFERENCES combos(id)   ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── CARGA HORÁRIA DOS BARBEIROS ───────────────────────────────
CREATE TABLE carga_horaria (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  barbeiro_id   INT UNSIGNED NOT NULL,
  dia_semana    TINYINT UNSIGNED NOT NULL, -- 0=Dom 1=Seg ... 6=Sáb
  hora_inicio   TIME NOT NULL,
  hora_fim      TIME NOT NULL,
  UNIQUE KEY uq_barbeiro_dia (barbeiro_id, dia_semana),
  FOREIGN KEY (barbeiro_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── BLOQUEIOS DE HORÁRIO ──────────────────────────────────────
CREATE TABLE bloqueios (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  barbeiro_id   INT UNSIGNED NOT NULL,
  data          DATE NOT NULL,
  hora_inicio   TIME NOT NULL,
  hora_fim      TIME NOT NULL,
  FOREIGN KEY (barbeiro_id) REFERENCES usuarios(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── AGENDAMENTOS ──────────────────────────────────────────────
CREATE TABLE agendamentos (
  id              INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  cliente_id      INT UNSIGNED NOT NULL,
  barbeiro_id     INT UNSIGNED NOT NULL,
  servico_id      INT UNSIGNED,              -- NULL se for combo
  combo_id        INT UNSIGNED,              -- NULL se for serviço
  data_hora       DATETIME NOT NULL,
  duracao_min     SMALLINT UNSIGNED NOT NULL,
  preco           DECIMAL(8,2) NOT NULL,     -- snapshot do preço no momento
  status          ENUM('pendente','confirmado','cancelado','nao_confirmado') NOT NULL DEFAULT 'pendente',
  token_confirm   VARCHAR(64) NOT NULL UNIQUE,
  confirmado_em   DATETIME,
  cancelado_em    DATETIME,
  cancelado_por   ENUM('cliente','barbeiro','admin'),
  criado_em       DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (cliente_id)  REFERENCES usuarios(id),
  FOREIGN KEY (barbeiro_id) REFERENCES usuarios(id),
  FOREIGN KEY (servico_id)  REFERENCES servicos(id),
  FOREIGN KEY (combo_id)    REFERENCES combos(id),
  CONSTRAINT chk_servico_ou_combo CHECK (
    (servico_id IS NOT NULL AND combo_id IS NULL) OR
    (servico_id IS NULL AND combo_id IS NOT NULL)
  )
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── PORTFÓLIO DOS BARBEIROS ───────────────────────────────────
CREATE TABLE portfolio (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  barbeiro_id   INT UNSIGNED NOT NULL,
  servico_id    INT UNSIGNED,               -- serviço associado à foto
  foto_path     VARCHAR(255) NOT NULL,
  legenda       VARCHAR(255),
  criado_em     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (barbeiro_id) REFERENCES usuarios(id) ON DELETE CASCADE,
  FOREIGN KEY (servico_id)  REFERENCES servicos(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── TICKETS DE SUGESTÃO DE SERVIÇO ───────────────────────────
CREATE TABLE tickets (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  barbeiro_id   INT UNSIGNED NOT NULL,
  nome          VARCHAR(100) NOT NULL,
  descricao     TEXT,
  categoria_id  INT UNSIGNED,
  preco         DECIMAL(8,2),
  duracao_min   SMALLINT UNSIGNED,
  status        ENUM('pendente','aprovado','recusado') NOT NULL DEFAULT 'pendente',
  obs_admin     TEXT,                       -- motivo de recusa ou nota
  criado_em     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  resolvido_em  DATETIME,
  FOREIGN KEY (barbeiro_id) REFERENCES usuarios(id),
  FOREIGN KEY (categoria_id) REFERENCES categorias(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- ── MENSAGENS RÁPIDAS ─────────────────────────────────────────
CREATE TABLE mensagens_rapidas (
  id            INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

  ativo         TINYINT(1) NOT NULL DEFAULT 1
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

<?php
session_start();

function exigir_login(): void {
  if (empty($_SESSION['usuario'])) {
    header('Location: /login.php');
    exit;
  }
}

function exigir_perfil(string ...$perfis): void {
  exigir_login();
  if (!in_array($_SESSION['usuario']['perfil'], $perfis, true)) {
    header('HTTP/1.1 403 Forbidden');
    header('Location: /login.php');
    exit;
  }
}

function usuario_logado(): ?array {
  return $_SESSION['usuario'] ?? null;
}

function eh_admin(): bool {
  return ($_SESSION['usuario']['perfil'] ?? '') === 'admin';
}

Geração de horários disponíveis

Este é o algoritmo central do projeto. Dado um barbeiro, uma data e uma duração, retorna os slots livres.

Regras que o algoritmo deve respeitar: carga horária do dia da semana · bloqueios manuais · agendamentos já existentes (status pendente ou confirmado) · duração do serviço/combo · antecedência máxima de 4 dias · não mostrar horários já passados no dia de hoje

Pseudocódigo — getSlots()

function getSlots(int $barbeiro_id, string $data, int $duracao_min): array {

  // 1. Verificar janela de 4 dias
  $hoje   = date('Y-m-d');
  $limite = date('Y-m-d', strtotime('+4 days'));
  if ($data < $hoje || $data > $limite) return [];

  // 2. Buscar carga horária do barbeiro para o dia da semana
  $dia_semana = (int) date('w', strtotime($data)); // 0=Dom
  $carga = query(
    "SELECT hora_inicio, hora_fim FROM carga_horaria
     WHERE barbeiro_id = ? AND dia_semana = ?",
    [$barbeiro_id, $dia_semana]
  );
  if (!$carga) return []; // barbeiro não trabalha esse dia

  // 3. Montar lista de slots de 45 em 45 min (ou duracao_min)
  $intervalo  = 45; // minutos entre início de cada slot
  $slots_brutos = [];
  $cursor = strtotime($data . ' ' . $carga['hora_inicio']);
  $fim    = strtotime($data . ' ' . $carga['hora_fim']);

  while (($cursor + $duracao_min * 60) <= $fim) {
    $slots_brutos[] = $cursor;
    $cursor += $intervalo * 60;
  }

  // 4. Buscar bloqueios do dia
  $bloqueios = queryAll(
    "SELECT hora_inicio, hora_fim FROM bloqueios
     WHERE barbeiro_id = ? AND data = ?",
    [$barbeiro_id, $data]
  );

  // 5. Buscar agendamentos já marcados no dia
  $agendados = queryAll(
    "SELECT data_hora, duracao_min FROM agendamentos
     WHERE barbeiro_id = ?
       AND DATE(data_hora) = ?
       AND status IN ('pendente','confirmado')",
    [$barbeiro_id, $data]
  );

  // 6. Filtrar slots conflitantes
  $agora = time();
  $slots_livres = [];

  foreach ($slots_brutos as $slot_ts) {
    // não mostrar no passado (hoje)
    if ($slot_ts <= $agora) continue;

    $slot_fim = $slot_ts + $duracao_min * 60;
    $conflito = false;

    // checar bloqueios
    foreach ($bloqueios as $b) {
      $b_ini = strtotime($data . ' ' . $b['hora_inicio']);
      $b_fim = strtotime($data . ' ' . $b['hora_fim']);
      if ($slot_ts < $b_fim && $slot_fim > $b_ini) {
        $conflito = true; break;
      }
    }
    if ($conflito) continue;

    // checar agendamentos
    foreach ($agendados as $ag) {
      $ag_ini = strtotime($ag['data_hora']);
      $ag_fim = $ag_ini + $ag['duracao_min'] * 60;
      if ($slot_ts < $ag_fim && $slot_fim > $ag_ini) {
        $conflito = true; break;
      }
    }
    if ($conflito) continue;

    $slots_livres[] = date('H:i', $slot_ts);
  }

  return $slots_livres;
}

<?php
require_once '/var/www/vendor/autoload.php';

use PHPMailer\PHPMailer\PHPMailer;
use PHPMailer\PHPMailer\SMTP;

function enviar_email(
  string $para_email,
  string $para_nome,
  string $assunto,
  string $corpo_html
): bool {
  $mail = new PHPMailer(true);
  try {
    $mail->isSMTP();
    $mail->Host       = $_ENV['MAIL_HOST'];
    $mail->SMTPAuth   = true;
    $mail->Username   = $_ENV['MAIL_USER'];
    $mail->Password   = $_ENV['MAIL_PASS'];
    $mail->SMTPSecure = PHPMailer::ENCRYPTION_STARTTLS;
    $mail->Port       = (int) $_ENV['MAIL_PORT'];
    $mail->CharSet    = 'UTF-8';

    $mail->setFrom($_ENV['MAIL_USER'], $_ENV['MAIL_FROM_NAME']);
    $mail->addAddress($para_email, $para_nome);

    $mail->isHTML(true);
    $mail->Subject = $assunto;
    $mail->Body    = $corpo_html;
    $mail->AltBody = strip_tags($corpo_html);

    $mail->send();
    return true;
  } catch (\Exception $e) {
    error_log('Erro de email: ' . $mail->ErrorInfo);
    return false;
  }
}

// ── helper: substitui variáveis no corpo de mensagens rápidas
function montar_mensagem(string $corpo, array $vars): string {
  foreach ($vars as $chave => $valor) {
    $corpo = str_replace('{' . $chave . '}', $valor, $corpo);
  }
  return $corpo;
}

Padrão de formulário POST

<?php
// todo form POST deve:
// 1. validar CSRF token
// 2. sanitizar inputs
// 3. usar prepared statements
// 4. redirecionar após sucesso (PRG pattern)

if ($_SERVER['REQUEST_METHOD'] === 'POST') {

  // CSRF
  if (!hash_equals($_SESSION['csrf_token'], $_POST['csrf_token'] ?? '')) {
    http_response_code(403);
    exit('Token inválido');
  }

  // sanitizar
  $nome = trim(htmlspecialchars($_POST['nome'] ?? '', ENT_QUOTES, 'UTF-8'));

  // prepared statement
  $stmt = $pdo->prepare("INSERT INTO ... VALUES (?)");
  $stmt->execute([$nome]);

  // PRG — evita resubmissão no F5
  header('Location: /destino.php?ok=1');
  exit;
}

// gerar CSRF token (incluir em todo form)
if (empty($_SESSION['csrf_token'])) {
  $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
}
